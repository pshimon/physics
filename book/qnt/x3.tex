\section*{A Minimal Axiomatic Core of Quantum Measurement}

\subsection*{Axiomatics}

\begin{axiom}[States]
A quantum system is described by a Hilbert space $\mathcal H$.
Physical states are positive trace-class operators $\rho$ on $\mathcal H$
with $\mathrm{Tr}\,\rho=1$.
\end{axiom}

\begin{axiom}[Measurements]
A measurement is specified by a normalized positive operator--valued measure (POVM)
\[
E:\mathcal B(\Omega)\to\mathcal B(\mathcal H),
\qquad E(\Omega)=\mathbb I.
\]
\end{axiom}

\begin{axiom}[Born rule]
For any state $\rho$ and measurable event $\Delta\subset\Omega$,
\[
\mathbb P(\Delta|\rho)=\mathrm{Tr}(\rho E(\Delta)).
\]
\end{axiom}

\begin{remark}
Operators do not represent measurements.
Measurements are event-valued objects.
\end{remark}

---

\subsection*{Sharp Measurements and Operators}

\begin{definition}[Sharp observable]
A measurement is sharp iff its POVM is projection-valued (PVM).
\end{definition}

\begin{definition}[Associated operator]
Given a PVM $E_A$, define the self-adjoint operator
\[
\hat A := \int_\Omega a\, dE_A(a).
\]
\end{definition}

\begin{proposition}
The operator $\hat A$ encodes only the moments of the probability measure
$\mu_\rho^A(\Delta)=\mathrm{Tr}(\rho E_A(\Delta))$.
\end{proposition}

\begin{corollary}
Operator action $\hat A\rho$ has no operational meaning as a measurement.
\end{corollary}

---

\subsection*{Definiteness and Eigenstates}

\begin{definition}[Definite value]
A state $\rho$ has a definite value of observable $A$ iff
\[
E_A(\Delta)\rho=\rho
\]
for some measurable $\Delta$.
\end{definition}

\begin{theorem}[No sharp states for continuous observables]
If $E_A(\{a\})=0$ for all $a\in\Omega$, then no normalizable state has a
definite value of $A$.
\end{theorem}

\begin{remark}
Generalized eigenstates (e.g.\ Dirac deltas) define the PVM but are not
physical states, as already implicit in the formalism of
\emph{:contentReference[oaicite:0]{index=0}}.
\end{remark}

---

\subsection*{Incompatibility and Noncommutativity}

\begin{definition}[Joint measurability]
Two observables $A,B$ are jointly sharply measurable iff there exists a
joint PVM $E_{AB}$ with marginals $E_A$ and $E_B$.
\end{definition}

\begin{theorem}[Compatibility criterion]
$A$ and $B$ admit a joint PVM iff
\[
[E_A(\Delta),E_B(\Gamma)]=0
\quad \forall \Delta,\Gamma.
\]
\end{theorem}

\begin{corollary}
If $[\hat A,\hat B]\neq 0$, then no joint sharp measurement exists.
\end{corollary}

\begin{remark}
Operator noncommutativity is a sufficient diagnostic for measurement
incompatibility, not its definition.
\end{remark}

---

\subsection*{Uncertainty as a Measurement-Theoretic Statement}

\begin{theorem}[Uncertainty principle]
There exists no POVM on phase space whose marginals are both arbitrarily
sharp position and momentum measurements.
\end{theorem}

\begin{corollary}
Heisenberg uncertainty expresses the nonexistence of a joint refinement of
the corresponding POVMs, not a limitation of operator algebra.
\end{corollary}

\begin{remark}
The usual variance inequality is a corollary of this structural fact,
clarifying the original insight of
\emph{:contentReference[oaicite:1]{index=1}}.
\end{remark}

---

\subsection*{State Update}

\begin{axiom}[Conditionalization]
Upon observing outcome $\Delta$, the state updates as
\[
\rho \mapsto
\frac{\sqrt{E(\Delta)}\,\rho\,\sqrt{E(\Delta)}}
{\mathrm{Tr}(\rho E(\Delta))}.
\]
\end{axiom}

\begin{theorem}
State update under measurement is not generated by any Hamiltonian and is
not a physical time evolution.
\end{theorem}

\begin{corollary}
Collapse is conditionalization on a non-Boolean event structure.
\end{corollary}

---

\subsection*{Summary}

\begin{itemize}
\item Measurements are POVMs/PVMs, not operators.
\item Operators summarize measurements but do not define them.
\item Noncommutativity means incompatibility of measurements.
\item Uncertainty is the absence of a joint POVM.
\item Collapse is information update, not dynamics.
\end{itemize}

\begin{center}
\boxed{\text{Quantum mechanics is probability theory on a non-Boolean event space.}}
\end{center}
